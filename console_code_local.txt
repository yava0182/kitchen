// ============================================
// PPDBG — единый DevTools-инжектор (вставить в консоль)
// Работает и на "нашей" странице, и на референсе, если движок уже загружен.
//
// После вставки используйте команды:
//   PPDBG.help()
//   PPDBG.status()
//   PPDBG.roomSizes.on()
//   PPDBG.module.create()
//   PPDBG.module.sizes()
//
// Важно: этот скрипт НЕ сохраняется в проект — это только для DevTools.
// ============================================

(function () {
  'use strict';

  const TAG = '[PPDBG]';
  const now = () => new Date().toISOString().slice(11, 19);
  const log = (...a) => console.log(TAG, now(), ...a);
  const warn = (...a) => console.warn(TAG, now(), ...a);
  const err = (...a) => console.error(TAG, now(), ...a);

  const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
  async function waitFor(check, { timeoutMs = 15000, intervalMs = 100, label = 'condition' } = {}) {
    const t0 = Date.now();
    while (Date.now() - t0 < timeoutMs) {
      try {
        if (check()) return true;
      } catch {}
      await sleep(intervalMs);
    }
    return false;
  }

  function getPP() {
    return window.pp || null;
  }
  function getRoom() {
    return window.room || window.pp?.three?.room || null;
  }
  function getScene() {
    return window.scene || window.pp?.three?.scene || null;
  }
  function getRenderer() {
    return window.renderer || window.pp?.three?.renderer || null;
  }
  function getCamera() {
    return window.camera || window.pp?.three?.camera || null;
  }
  function getDObjects() {
    return window.d_objects || null;
  }
  function getSc() {
    // sc/scene_controller/global_options in engine are often declared via top-level `let`
    // and therefore are NOT available on window.*. Try eval first, then window.*.
    try {
      // direct eval in DevTools can see global lexical bindings on the page
      // (works for engine bundles like main_core_v2.js)
      // eslint-disable-next-line no-eval
      const v = eval(
        '(typeof sc !== "undefined" && sc) ? sc : ((typeof scene_controller !== "undefined" && scene_controller) ? scene_controller : null)'
      );
      if (v) return v;
    } catch {}
    return window.sc || window.scene_controller || null;
  }
  function getGlobalOptions() {
    try {
      // eslint-disable-next-line no-eval
      return eval('typeof global_options !== "undefined" ? global_options : null');
    } catch {}
    return window.global_options || null;
  }
  function patchGlobalOptions(patch) {
    const go = getGlobalOptions();
    if (!go) return false;
    try {
      Object.keys(patch || {}).forEach((k) => {
        go[k] = patch[k];
      });
      return true;
    } catch {
      return false;
    }
  }
  function render() {
    try {
      if (window.pp && typeof window.pp.render === 'function') return window.pp.render();
      if (window.pp?.three && typeof window.pp.three.render === 'function') return window.pp.three.render();
    } catch (e) {
      warn('render failed', e);
    }
  }
  function forceRenderOnce() {
    try {
      const r = getRenderer();
      const s = getScene();
      const c = getCamera();
      if (r && s && c && typeof r.render === 'function') {
        r.render(s, c);
        return true;
      }
    } catch (e) {
      warn('forceRenderOnce failed', e);
    }
    return false;
  }

  // --- Safe wrapper: single_preview should never leave room hidden ---
  const SINGLE_PREVIEW_WRAP_KEY = '__ppdbg_single_preview_original__';
  function safeWrapSinglePreview(enable) {
    const sp = window.single_preview;
    if (enable) {
      if (typeof sp !== 'function') {
        warn('single_preview not found');
        return false;
      }
      if (window[SINGLE_PREVIEW_WRAP_KEY]) return true;
      window[SINGLE_PREVIEW_WRAP_KEY] = sp;
      window.single_preview = function (flag, obj) {
        const room = getRoom();
        try {
          return window[SINGLE_PREVIEW_WRAP_KEY].apply(this, arguments);
        } catch (e) {
          err('single_preview crashed; restoring room visibility', e);
          try {
            if (room?.show) room.show();
            if (room) room.visible = true;
          } catch {}
          throw e;
        }
      };
      log('safe single_preview wrapper ENABLED');
      return true;
    } else {
      if (!window[SINGLE_PREVIEW_WRAP_KEY]) return true;
      window.single_preview = window[SINGLE_PREVIEW_WRAP_KEY];
      window[SINGLE_PREVIEW_WRAP_KEY] = null;
      log('safe single_preview wrapper DISABLED');
      return true;
    }
  }

  // --- Module helpers ---
  function clone(obj) {
    // copy_object exists in engine; fallback to structuredClone
    try {
      if (typeof window.copy_object === 'function') return window.copy_object(obj);
    } catch {}
    try {
      if (typeof structuredClone === 'function') return structuredClone(obj);
    } catch {}
    return JSON.parse(JSON.stringify(obj));
  }

  function mergeDeep(dst, src) {
    // lodash is available as _ in both reference & our build usually
    try {
      if (window._ && typeof _.merge === 'function') return _.merge(dst, src);
    } catch {}
    // minimal merge fallback
    const isObj = (v) => v && typeof v === 'object' && !Array.isArray(v);
    Object.keys(src || {}).forEach((k) => {
      if (isObj(dst[k]) && isObj(src[k])) mergeDeep(dst[k], src[k]);
      else dst[k] = src[k];
    });
    return dst;
  }

  function countCabinetSizes(cab) {
    const own = cab?.userData?.own_object;
    if (!own?.children) return 0;
    return own.children.filter((c) => c?.constructor?.name === 'CabinetSize' || c?.userData?.class === 'CabinetSize').length;
  }

  function ensureRoomShown() {
    const room = getRoom();
    try {
      if (room?.show) room.show();
      if (room) room.visible = true;
    } catch {}
  }

  // --- Public API ---
  const api = (window.PPDBG = window.PPDBG || {});
  api.version = '2026-01-21-devtools-v1';

  api.help = function () {
    log('Commands:');
    console.log(`
${TAG} ${api.version}
PPDBG.status()
PPDBG.waitEngine()
PPDBG.safeSinglePreview.on() / PPDBG.safeSinglePreview.off()
PPDBG.roomSizes.on() / PPDBG.roomSizes.off()
PPDBG.module.create([overrides])
PPDBG.module.sizes([cabinet])
PPDBG.module.previewOn([cabinet]) / PPDBG.module.previewOff([cabinet])
`);
  };

  api.status = function () {
    const pp = getPP();
    const room = getRoom();
    const scene = getScene();
    const sc = getSc();
    const d = getDObjects();
    const go = getGlobalOptions();
    const renderer = getRenderer();
    const camera = getCamera();
    const canvas = renderer?.domElement || null;
    const status = {
      has_pp: !!pp,
      has_room: !!room,
      has_scene: !!scene,
      has_sc_add_object: !!(sc && typeof sc.add_object === 'function'),
      d_objects_len: Array.isArray(d) ? d.length : null,
      room_sizes_children: room?.sizes?.children?.length ?? null,
      common_sizes_children: room?.common_sizes?.children?.length ?? null,
      room_visible: room?.visible ?? null,
      renderer: renderer
        ? {
            has_canvas: !!canvas,
            canvas_size: canvas ? { w: canvas.width, h: canvas.height } : null,
            client_size: canvas ? { w: canvas.clientWidth, h: canvas.clientHeight } : null,
            in_dom: canvas ? document.contains(canvas) : null
          }
        : null,
      camera: camera
        ? {
            pos: camera.position ? { x: camera.position.x, y: camera.position.y, z: camera.position.z } : null,
            fov: camera.fov,
            near: camera.near,
            far: camera.far
          }
        : null,
      global_options: go ? {
        common_sizes_visible: go.common_sizes_visible,
        sizes_visible: go.sizes_visible,
        single_preview: go.single_preview
      } : null
    };
    log('status', status);
    return status;
  };

  api.forceRender = function () {
    const ok = forceRenderOnce();
    log('forceRender', ok ? 'OK' : 'FAILED');
    return ok;
  };

  const LOOP_KEY = '__ppdbg_render_loop__';
  api.renderLoop = {
    start() {
      if (window[LOOP_KEY]) return true;
      let alive = true;
      const tick = () => {
        if (!alive) return;
        forceRenderOnce();
        requestAnimationFrame(tick);
      };
      window[LOOP_KEY] = () => {
        alive = false;
      };
      tick();
      log('renderLoop started');
      return true;
    },
    stop() {
      if (!window[LOOP_KEY]) return true;
      try {
        window[LOOP_KEY]();
      } catch {}
      window[LOOP_KEY] = null;
      log('renderLoop stopped');
      return true;
    }
  };

  const CUBE_KEY = '__ppdbg_debug_cube__';
  api.debugCube = {
    on() {
      try {
        const scene = getScene();
        if (!scene) {
          warn('debugCube.on: scene not ready');
          return false;
        }
        if (window[CUBE_KEY]) return true;
        const geom = new THREE.BoxGeometry(1, 1, 1);
        const mat = new THREE.MeshNormalMaterial();
        const cube = new THREE.Mesh(geom, mat);
        cube.position.set(0, 1, 0);
        cube.name = 'PPDBG_DebugCube';
        scene.add(cube);
        window[CUBE_KEY] = cube;
        log('debugCube added');
        forceRenderOnce();
        return true;
      } catch (e) {
        err('debugCube.on failed', e);
        return false;
      }
    },
    off() {
      try {
        const cube = window[CUBE_KEY];
        if (!cube) return true;
        cube.parent?.remove(cube);
        cube.geometry?.dispose?.();
        cube.material?.dispose?.();
        window[CUBE_KEY] = null;
        log('debugCube removed');
        forceRenderOnce();
        return true;
      } catch (e) {
        err('debugCube.off failed', e);
        return false;
      }
    }
  };

  api.waitEngine = async function (opts = {}) {
    const ok = await waitFor(
      () => !!getPP() && !!getRoom() && (!!window.build_sizes_all || (getSc() && typeof getSc().add_object === 'function')),
      { timeoutMs: opts.timeoutMs ?? 20000, intervalMs: 100, label: 'engine' }
    );
    log(ok ? 'engine ready' : 'engine NOT ready (timeout)');
    return ok;
  };

  api.safeSinglePreview = {
    on() {
      return safeWrapSinglePreview(true);
    },
    off() {
      return safeWrapSinglePreview(false);
    }
  };

  api.roomSizes = {
    on() {
      ensureRoomShown();
      // Если ранее single_preview упал и оставил комнату скрытой — принудительно выключаем режим.
      patchGlobalOptions({ single_preview: false, common_sizes_visible: true, sizes_visible: true });
      const room = getRoom();

      // 1) БАЗОВЫЕ стрелки комнаты лежат в room.sizes (создаются в Room.build_sizes()) — включаем их!
      if (room?.sizes) {
        room.sizes.visible = true;
        if (Array.isArray(room.sizes.children)) {
          room.sizes.children.forEach((c) => {
            if (c) c.visible = true;
          });
        }
      }

      // 2) Сегментные/доп. стрелки комнаты (по объектам) лежат в room.common_sizes и создаются build_sizes_all()
      if (room?.common_sizes) {
        room.common_sizes.visible = true;
        // Иногда children остаются невидимыми после hide/show
        if (Array.isArray(room.common_sizes.children)) {
          room.common_sizes.children.forEach((c) => {
            if (c) c.visible = true;
          });
        }
      }
      if (typeof window.build_sizes_all === 'function') {
        const before = room?.common_sizes?.children?.length ?? null;
        log('roomSizes.on → build_sizes_all()', { before_common_children: before, room_sizes_children: room?.sizes?.children?.length ?? null });
        try {
          window.build_sizes_all();
        } catch (e) {
          err('build_sizes_all failed', e);
        }
        const after = room?.common_sizes?.children?.length ?? null;
        log('roomSizes.on → build_sizes_all() done', { after_common_children: after, room_sizes_children: room?.sizes?.children?.length ?? null });
      } else {
        warn('build_sizes_all not found');
      }
      render();
      return true;
    },
    off() {
      patchGlobalOptions({ common_sizes_visible: false });
      const room = getRoom();
      if (room?.common_sizes) room.common_sizes.visible = false;
      render();
      return true;
    }
  };

  api.module = {
    create(overrides = {}) {
      const sc = getSc();
      if (!sc || typeof sc.add_object !== 'function') {
        warn('module.create: sc.add_object not ready');
        return null;
      }
      const pp = getPP();
      const base = pp?.engine?.item_def_params?.module ? clone(pp.engine.item_def_params.module) : null;
      if (!base) {
        warn('module.create: pp.engine.item_def_params.module not ready; using minimal params');
      }
      const params = base || {
        type: 'module',
        size: { x: 300, y: 400, z: 200 },
        spec: { cabinet_group: 'bottom' },
        variables: { visible: 1, main: {} },
        children: [],
        holes: []
      };

      // make it deterministic and allowed for _build_sizes
      mergeDeep(params, {
        type: 'module',
        size: { x: 300, y: 400, z: 200 },
        spec: { cabinet_group: 'bottom', hide_sizes: 0, is_fill: 0 }
      });
      mergeDeep(params, overrides || {});

      // IMPORTANT: third param must be false in embedded/no-UI mode
      log('module.create → sc.add_object("module", params, false)', { size: params.size, cabinet_group: params.spec?.cabinet_group });
      let cab = null;
      try {
        cab = sc.add_object('module', params, false);
      } catch (e) {
        err('module.create failed', e);
        return null;
      }
      if (!cab) {
        err('module.create: sc.add_object returned null');
        return null;
      }

      try {
        cab.userData = cab.userData || {};
        cab.userData.is_root = true;
        if (cab.position?.set) cab.position.set(100, 0, 100);
        window.cabinet = cab;
        try { eval('cabinet = window.cabinet;'); } catch {}
      } catch {}

      render();
      log('module.create OK', { name: cab.name, has_build_sizes: typeof cab._build_sizes === 'function', d_objects_len: (getDObjects() || []).length });
      return cab;
    },

    sizes(cabinetObj) {
      const cab = cabinetObj || window.cabinet || null;
      if (!cab) {
        warn('module.sizes: cabinet not found (create it first: PPDBG.module.create())');
        return null;
      }
      if (typeof cab._build_sizes !== 'function') {
        warn('module.sizes: cab._build_sizes not found');
        return null;
      }
      // make sure flags allow sizes
      try {
        cab.userData = cab.userData || {};
        cab.userData.is_root = true;
        cab.userData.params = cab.userData.params || {};
        cab.userData.params.spec = cab.userData.params.spec || {};
        cab.userData.params.spec.hide_sizes = 0;
        cab.userData.params.spec.is_fill = 0;
      } catch {}

      log('module.sizes → cab._build_sizes()');
      try {
        cab._build_sizes();
      } catch (e) {
        err('cab._build_sizes failed', e);
        return null;
      }

      // force visibility
      try {
        const own = cab.userData?.own_object;
        if (own) own.visible = true;
        if (own?.children) own.children.forEach((c) => (c.visible = true));
      } catch {}

      const n = countCabinetSizes(cab);
      log('module.sizes OK', { CabinetSize: n });
      render();
      return n;
    },

    previewOn(cabinetObj) {
      api.safeSinglePreview.on();
      const cab = cabinetObj || window.cabinet || null;
      if (!cab) {
        warn('module.previewOn: cabinet not found');
        return false;
      }
      if (typeof window.single_preview !== 'function') {
        warn('module.previewOn: single_preview not found');
        return false;
      }
      try {
        log('module.previewOn → single_preview(true, cabinet)');
        window.single_preview(true, cab);
        render();
        return true;
      } catch (e) {
        err('module.previewOn failed', e);
        return false;
      }
    },

    previewOff(cabinetObj) {
      api.safeSinglePreview.on();
      const cab = cabinetObj || window.cabinet || null;
      if (!cab) {
        warn('module.previewOff: cabinet not found');
        return false;
      }
      if (typeof window.single_preview !== 'function') {
        warn('module.previewOff: single_preview not found');
        return false;
      }
      try {
        log('module.previewOff → single_preview(false, cabinet)');
        window.single_preview(false, cab);
        ensureRoomShown();
        render();
        return true;
      } catch (e) {
        err('module.previewOff failed', e);
        ensureRoomShown();
        return false;
      }
    }
  };

  // default safety: avoid "gray scene" from crashes
  api.safeSinglePreview.on();

  log('installed', api.version);
  api.help();
})();
